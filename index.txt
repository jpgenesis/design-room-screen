<script>
/* Design Room dashboard. Edit room.json to change priorities and images. */
(function () {
  var HOME = 'America/Los_Angeles';
  var DEFAULTS = {
    studios: [
      { name: 'San Carlos', tz: 'America/Los_Angeles' },
      { name: 'London', tz: 'Europe/London' },
      { name: 'Paris', tz: 'Europe/Paris' }
    ],
    workday: [9, 18],
    weather: { lat: 37.5072, lon: -122.2605, place: 'San Carlos' },
    priorities: [
      { area: 'UX', text: 'Design system across every product' },
      { area: 'Product', text: 'Fleet Management Platform, phase 1' },
      { area: 'Brand', text: 'genesis.ai fixes: type, footer, favicon' }
    ],
    images: [],
    imageEveryHours: 24
  };
  var SAMPLE_WX = { f: 58, text: 'Partly sunny', hi: 63, lo: 47, sample: true };
  var cfg = DEFAULTS, wx = null;

  function el(tag, cls, text) { var n = document.createElement(tag); if (cls) n.className = cls; if (text != null) n.textContent = text; return n; }
  function parts(tz, d) {
    var o = {};
    new Intl.DateTimeFormat('en-GB', { timeZone: tz, hourCycle: 'h23', hour: '2-digit', minute: '2-digit', weekday: 'short' })
      .formatToParts(d).forEach(function (p) { o[p.type] = p.value; });
    o.h = parseInt(o.hour, 10) + parseInt(o.minute, 10) / 60;
    return o;
  }
  function hhmm(h) { h = ((h % 24) + 24) % 24; var H = Math.floor(h), M = Math.round((h - H) * 60); if (M === 60) { H++; M = 0; } return String(H % 24).padStart(2, '0') + ':' + String(M).padStart(2, '0'); }
  function weekend(day) { return day === 'Sat' || day === 'Sun'; }
  function segs(a, len) { a = ((a % 24) + 24) % 24; var b = a + len; return b > 24 ? [[a, 24], [0, b - 24]] : [[a, b]]; }
  function inter(x, y) { var out = []; x.forEach(function (p) { y.forEach(function (q) { var s = Math.max(p[0], q[0]), e = Math.min(p[1], q[1]); if (e > s) out.push([s, e]); }); }); return out; }

  function studiosBlock(now) {
    var home = parts(HOME, now), wd = cfg.workday, len = wd[1] - wd[0];
    var box = el('div', 'studios');
    box.appendChild(el('div', 'lab', 'Studios, San Carlos time'));
    var tracks = el('div', 'tracks');
    var homeSeg = [[wd[0], wd[1]]];
    var lines = [];
    cfg.studios.forEach(function (s, i) {
      var p = parts(s.tz, now);
      var diff = Math.round(((((p.h - home.h) % 24) + 24) % 24) * 4) / 4;
      if (diff > 12) diff -= 24;
      var start = wd[0] - diff;
      var mine = segs(start, len);
      var row = el('div', 'srow');
      var n = el('span', 'n', s.name); if (s.note) n.appendChild(el('em', null, s.note));
      var t = el('span', 't'); if (p.weekday !== home.weekday) t.appendChild(el('em', null, p.weekday)); t.appendChild(document.createTextNode(p.hour + ':' + p.minute));
      row.appendChild(n); row.appendChild(t);
      var trk = el('div', 'trk');
      var ov = i === 0 ? [] : inter(mine, homeSeg);
      mine.forEach(function (r) { var b = el('i', i === 0 ? 'home' : ''); b.style.left = r[0] / 24 * 100 + '%'; b.style.width = (r[1] - r[0]) / 24 * 100 + '%'; trk.appendChild(b); });
      var homeWeekend = weekend(home.weekday);
      ov.forEach(function (r) {
        var localDay = parts(s.tz, new Date(now.getTime() + (r[0] - home.h) * 3600000)).weekday;
        if (homeWeekend || weekend(localDay)) return;
        var b = el('i', 'ov'); b.style.left = r[0] / 24 * 100 + '%'; b.style.width = (r[1] - r[0]) / 24 * 100 + '%'; trk.appendChild(b);
        lines.push([s.name, r]);
      });
      var nk = el('b', 'nowtick'); nk.style.left = home.h / 24 * 100 + '%'; trk.appendChild(nk);
      row.appendChild(trk);
      tracks.appendChild(row);
    });
    var axis = el('div', 'axis');
    [0, 6, 12, 18, 24].forEach(function (h) { var a = el('span', null, h === 24 ? '24' : String(h).padStart(2, '0')); a.style.left = h / 24 * 100 + '%'; axis.appendChild(a); });
    tracks.appendChild(axis);
    box.appendChild(tracks);
    var ovText = el('div', 'overlap');
    var key = el('span', 'key'); key.appendChild(el('i')); key.appendChild(document.createTextNode('Shared hours with San Carlos: '));
    ovText.appendChild(key);
    var bits = lines.filter(function (l) { return l[1]; });
    if (!bits.length) ovText.appendChild(document.createTextNode('none today.'));
    bits.forEach(function (l, j) {
      ovText.appendChild(el('b', null, l[0]));
      ovText.appendChild(document.createTextNode(' ' + hhmm(l[1][0]) + ' to ' + hhmm(l[1][1]) + (j < bits.length - 1 ? ', ' : '.')));
    });
    box.appendChild(ovText);
    return box;
  }

  function render(root) {
    var now = new Date();
    var panel = root.querySelector('.panel');
    panel.textContent = '';
    var home = parts(HOME, now);
    var top = el('div', 'now');
    var left = el('div');
    left.appendChild(el('div', 'clock', home.hour + ':' + home.minute));
    left.appendChild(el('div', 'date', new Intl.DateTimeFormat('en-GB', { timeZone: HOME, weekday: 'long', day: 'numeric', month: 'long' }).format(now)));
    top.appendChild(left);
    var w = wx || SAMPLE_WX, r = el('div', 'wx');
    var temp = el('div', 'temp', w.f + '°'); temp.appendChild(el('small', null, 'F, ' + Math.round((w.f - 32) * 5 / 9) + '°C'));
    r.appendChild(temp);
    r.appendChild(el('div', 'cond', w.text + (w.hi != null ? ', high ' + w.hi + '°, low ' + w.lo + '°' : '')));
    top.appendChild(r);
    panel.appendChild(top);
    panel.appendChild(studiosBlock(now));
    var pr = el('div', 'prio');
    pr.appendChild(el('div', 'lab head', 'Design team priorities'));
    cfg.priorities.slice(0, 3).forEach(function (p) {
      var d = el('div'); d.appendChild(el('div', 'k', p.area)); d.appendChild(el('div', 'v', p.text)); pr.appendChild(d);
    });
    panel.appendChild(pr);
  }

  function sampleImage(bg) {
    var c = document.createElement('canvas'), W = 900, H = 900; c.width = W; c.height = H;
    var g = c.getContext('2d');
    var lg = g.createLinearGradient(0, 0, W, H); lg.addColorStop(0, '#E8E4DC'); lg.addColorStop(.55, '#C9CCC6'); lg.addColorStop(1, '#8E979A');
    g.fillStyle = lg; g.fillRect(0, 0, W, H);
    g.fillStyle = 'rgba(255,255,255,.35)'; g.beginPath(); g.arc(W * .72, H * .3, W * .26, 0, Math.PI * 2); g.fill();
    g.fillStyle = 'rgba(60,66,70,.35)'; g.fillRect(W * .08, H * .52, W * .34, H * .48);
    g.fillStyle = 'rgba(29,32,55,.22)'; g.fillRect(W * .42, H * .64, W * .5, H * .36);
    bg.appendChild(c);
  }
  function setImage(root) {
    var bg = root.querySelector('.bg'), credit = root.querySelector('.credit');
    var imgs = cfg.images || [];
    if (!imgs.length) { if (!bg.firstChild) sampleImage(bg); credit.textContent = 'Sample image. Your inspiration image goes here.'; return; }
    var slot = Math.floor(Date.now() / (cfg.imageEveryHours * 3600000)) % imgs.length;
    var im = imgs[slot];
    bg.textContent = ''; bg.style.backgroundImage = 'url("' + im.src + '")';
    credit.textContent = im.credit || '';
  }

  function loadConfig() {
    return fetch('room.json', { cache: 'no-store' }).then(function (r) { if (!r.ok) throw 0; return r.json(); })
      .then(function (j) { var c = {}; Object.keys(DEFAULTS).forEach(function (k) { c[k] = j[k] != null ? j[k] : DEFAULTS[k]; }); cfg = c; })
      .catch(function () { cfg = DEFAULTS; });
  }
  function loadWeather() {
    var p = cfg.weather;
    function get(u) { return fetch(u).then(function (r) { if (!r.ok) throw 0; return r.json(); }); }
    return get('https://api.weather.gov/points/' + p.lat + ',' + p.lon).then(function (pt) {
      return Promise.all([get(pt.properties.forecast), get(pt.properties.forecastHourly)]);
    }).then(function (res) {
      var ps = res[0].properties.periods, a = ps[0], b = ps[1] || a;
      var n = res[1].properties.periods[0];
      wx = { f: n.temperature, text: n.shortForecast, hi: a.isDaytime ? a.temperature : b.temperature, lo: a.isDaytime ? b.temperature : a.temperature };
    }).catch(function () { /* keep the last reading, or the sample in a preview */ });
  }

  function start() {
    var roots = Array.prototype.slice.call(document.querySelectorAll('.dash'));
    function all() { roots.forEach(render); }
    function imgs() { roots.forEach(setImage); }
    loadConfig().then(function () { imgs(); all(); return loadWeather(); }).then(all);
    setInterval(all, 20000);
    setInterval(function () { loadConfig().then(loadWeather).then(function () { imgs(); all(); }); }, 15 * 60000);
    var k = 0;
    setInterval(function () { k = (k + 1) % 4; var dx = [0, 3, 3, 0][k], dy = [0, 0, 3, 3][k];
      roots.forEach(function (r) { r.querySelector('.shift').style.transform = 'translate(' + dx + 'px,' + dy + 'px)'; }); }, 4 * 60000);
    if (document.body.hasAttribute('data-device')) {
      setTimeout(function () { location.reload(); }, 6 * 3600000);
      try { navigator.wakeLock && navigator.wakeLock.request('screen').catch(function () {}); } catch (e) {}
      document.addEventListener('visibilitychange', function () { try { if (!document.hidden && navigator.wakeLock) navigator.wakeLock.request('screen').catch(function () {}); } catch (e) {} });
    }
  }
  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', start); else start();
})();

</script>
